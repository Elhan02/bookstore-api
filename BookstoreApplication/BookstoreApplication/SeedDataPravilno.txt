public static class SeedData
{
    public static async Task InitializeAsync(IServiceProvider serviceProvider)
    {
        var roleManager = serviceProvider.GetRequiredService<RoleManager<IdentityRole>>();
        var userManager = serviceProvider.GetRequiredService<UserManager<ApplicationUser>>();

        string[] roles = { "Administrator", "Nutritionist", "Caretaker" };
        foreach (var role in roles)
        {
            if (!await roleManager.RoleExistsAsync(role))
            {
                await roleManager.CreateAsync(new IdentityRole(role));
            }
        }

        // ADMIN 1
        var admin1 = await userManager.FindByNameAsync("john");
        if (admin1 == null)
        {
            admin1 = new ApplicationUser
            {
                UserName = "john",
                Email = "john.doe@example.com",
                Name = "John",
                Surname = "Doe",
                EmailConfirmed = true
            };

            await userManager.CreateAsync(admin1, "John123!");
        }

        if (!await userManager.IsInRoleAsync(admin1, "Administrator"))
        {
            await userManager.AddToRoleAsync(admin1, "Administrator");
        }


        // ADMIN 2
        var admin2 = await userManager.FindByNameAsync("jane");
        if (admin2 == null)
        {
            admin2 = new ApplicationUser
            {
                UserName = "jane",
                Email = "jane.doe@example.com",
                Name = "Jane",
                Surname = "Doe",
                EmailConfirmed = true
            };

            await userManager.CreateAsync(admin2, "Jane123!");
        }

        if (!await userManager.IsInRoleAsync(admin2, "Administrator"))
        {
            await userManager.AddToRoleAsync(admin2, "Administrator");
        }


        // NUTRITIONIST
        var nutritionist1 = await userManager.FindByNameAsync("michael");
        if (nutritionist1 == null)
        {
            nutritionist1 = new ApplicationUser
            {
                UserName = "michael",
                Email = "michael.smith@example.com",
                Name = "Michael",
                Surname = "Smith",
                EmailConfirmed = true
            };

            await userManager.CreateAsync(nutritionist1, "Michael123!");
        }

        if (!await userManager.IsInRoleAsync(nutritionist1, "Nutritionist"))
        {
            await userManager.AddToRoleAsync(nutritionist1, "Nutritionist");
        }


        // CARETAKER
        var caretaker1 = await userManager.FindByNameAsync("sean");
        if (caretaker1 == null)
        {
            caretaker1 = new ApplicationUser
            {
                UserName = "sean",
                Email = "sean.buttler@example.com",
                Name = "Sean",
                Surname = "Buttler",
                EmailConfirmed = true
            };

            await userManager.CreateAsync(caretaker1, "Sean123!");
        }

        if (!await userManager.IsInRoleAsync(caretaker1, "Caretaker"))
        {
            await userManager.AddToRoleAsync(caretaker1, "Caretaker");
        }


    }
}
